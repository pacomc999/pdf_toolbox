# PDF Merge Tool Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-file browser tool that lets the user pick multiple PDFs, drag to reorder them, and download a merged PDF.

**Architecture:** Single `index.html` with all HTML, CSS, and JS inline. `pdf-lib` loaded from CDN handles all PDF manipulation. No server, no upload, everything runs locally in the browser.

**Tech Stack:** HTML5, vanilla JS, CSS, pdf-lib 1.17.1 (CDN), HTML5 Drag and Drop API, File API, Blob API.

```js
// ---- State ----
// files is an array of { file: File, error: string|null }
let files = [];

// ---- Element refs ----
const dropZone   = document.getElementById('drop-zone');
const fileInput  = document.getElementById('file-input');
const pickLabel  = document.getElementById('pick-label');
const fileList   = document.getElementById('file-list');
const actions    = document.getElementById('actions');
const mergeBtn   = document.getElementById('merge-btn');
const statusEl   = document.getElementById('status');

// ---- File picking ----
pickLabel.addEventListener('click', () => fileInput.click());
dropZone.addEventListener('click', (e) => {
  if (e.target !== pickLabel) fileInput.click();
});

fileInput.addEventListener('change', () => {
  addFiles(Array.from(fileInput.files));
  fileInput.value = ''; // allow re-picking same file
});

// ---- Drag-over drop zone (file OS drag) ----
dropZone.addEventListener('dragover', (e) => {
  e.preventDefault();
  dropZone.classList.add('drag-over');
});
dropZone.addEventListener('dragleave', () => dropZone.classList.remove('drag-over'));
dropZone.addEventListener('drop', (e) => {
  e.preventDefault();
  dropZone.classList.remove('drag-over');
  const dropped = Array.from(e.dataTransfer.files).filter(f => f.type === 'application/pdf' || f.name.endsWith('.pdf'));
  addFiles(dropped);
});

// ---- Add files to state ----
function addFiles(newFiles) {
  newFiles.forEach(f => {
    // Avoid exact duplicate filenames already in list
    const alreadyAdded = files.some(entry => entry.file.name === f.name && entry.file.size === f.size);
    if (!alreadyAdded) files.push({ file: f, error: null });
  });
  render();
}

// ---- Remove a file ----
function removeFile(index) {
  files.splice(index, 1);
  render();
}

// ---- Render file list ----
function render() {
  fileList.innerHTML = '';

  files.forEach((entry, index) => {
    const li = document.createElement('li');
    li.className = 'file-item';
    li.dataset.index = index;
    li.draggable = true;

    li.innerHTML = `
      <span class="drag-handle">&#9776;</span>
      <span class="file-name" title="${entry.file.name}">${entry.file.name}</span>
      ${entry.error ? `<span class="file-error">${entry.error}</span>` : ''}
      <button class="remove-btn" aria-label="Remove">&times;</button>
    `;

    li.querySelector('.remove-btn').addEventListener('click', () => removeFile(index));
    fileList.appendChild(li);
  });

  // Show/hide merge button area
  actions.classList.toggle('visible', files.length >= 2);
  statusEl.textContent = '';
}
```

```js
// ---- Drag-to-reorder handlers ----
let dragSrcIndex = null;

function onDragStart(e) {
  dragSrcIndex = parseInt(e.currentTarget.dataset.index);
  e.currentTarget.classList.add('dragging');
  e.dataTransfer.effectAllowed = 'move';
  // Required for Firefox
  e.dataTransfer.setData('text/plain', dragSrcIndex);
}

function onDragOver(e) {
  e.preventDefault();
  e.dataTransfer.dropEffect = 'move';
  e.currentTarget.classList.add('drag-target');
}

function onDragLeave(e) {
  e.currentTarget.classList.remove('drag-target');
}

function onDrop(e) {
  e.preventDefault();
  e.stopPropagation();
  const targetIndex = parseInt(e.currentTarget.dataset.index);
  if (dragSrcIndex === null || dragSrcIndex === targetIndex) return;

  // Move item in array
  const moved = files.splice(dragSrcIndex, 1)[0];
  files.splice(targetIndex, 0, moved);
  render();
}

function onDragEnd(e) {
  e.currentTarget.classList.remove('dragging');
  // Clean up any leftover drag-target highlights
  document.querySelectorAll('.drag-target').forEach(el => el.classList.remove('drag-target'));
  dragSrcIndex = null;
}
```

- [ ] **Step 1: Add the `mergePDFs()` function after the drag handler functions**

```js
// ---- Merge and download ----
mergeBtn.addEventListener('click', mergePDFs);

async function mergePDFs() {
  mergeBtn.disabled = true;
  statusEl.textContent = 'Merging...';

  // Clear previous errors
  files.forEach(entry => entry.error = null);
  render();
  mergeBtn.disabled = true; // re-disable after render

  try {
    const { PDFDocument } = PDFLib;
    const merged = await PDFDocument.create();

    for (const entry of files) {
      // Read file as ArrayBuffer
      const arrayBuffer = await readFileAsArrayBuffer(entry.file);

      let srcDoc;
      try {
        srcDoc = await PDFDocument.load(arrayBuffer);
      } catch (err) {
        entry.error = 'Not a valid PDF';
        continue; // skip this file, keep going
      }

      const pageIndices = srcDoc.getPageIndices();
      const copiedPages = await merged.copyPages(srcDoc, pageIndices);
      copiedPages.forEach(page => merged.addPage(page));
    }

    // If every file had an error, bail out
    const allFailed = files.every(entry => entry.error !== null);
    if (allFailed) {
      statusEl.textContent = 'No valid PDFs to merge.';
      render();
      mergeBtn.disabled = false;
      return;
    }

    // Check if any had errors to warn user
    const anyFailed = files.some(entry => entry.error !== null);

    const pdfBytes = await merged.save();
    downloadBlob(pdfBytes, 'merged.pdf');

    statusEl.textContent = anyFailed
      ? 'Done. Some files were skipped (see errors above).'
      : 'Done! Your merged PDF has been downloaded.';

    render();
    mergeBtn.disabled = false;

  } catch (err) {
    statusEl.textContent = 'Something went wrong. Please try again.';
    mergeBtn.disabled = false;
  }
}

// ---- Helper: read File as ArrayBuffer ----
function readFileAsArrayBuffer(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload  = () => resolve(reader.result);
    reader.onerror = () => reject(reader.error);
    reader.readAsArrayBuffer(file);
  });
}

// ---- Helper: trigger file download ----
function downloadBlob(bytes, filename) {
  const blob = new Blob([bytes], { type: 'application/pdf' });
  const url  = URL.createObjectURL(blob);
  const a    = document.createElement('a');
  a.href     = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}
```
